import streamlit as st
import numpy as np
import pandas as pd

st.set_page_config(page_title="Simulador Pro de Fútbol", page_icon="⚽", layout="centered")

st.title("⚽ Simulador Estadístico Monte Carlo")
st.caption("Predicción basada en Distribución de Poisson + 10,000 simulaciones")

# Base de datos preinstalada de equipos
equipos_db = {
    "Inter Miami CF": {"gf": 2.20, "gc": 1.25, "gf1t": 1.10, "corners": 4.8},
    "LA Galaxy": {"gf": 1.90, "gc": 1.40, "gf1t": 0.80, "corners": 5.1},
    "Real Madrid": {"gf": 2.15, "gc": 0.80, "gf1t": 1.00, "corners": 5.8},
    "FC Barcelona": {"gf": 2.05, "gc": 0.95, "gf1t": 0.90, "corners": 6.2},
    "Chelsea FC": {"gf": 1.75, "gc": 1.30, "gf1t": 0.80, "corners": 5.4},
    "Juventus": {"gf": 1.45, "gc": 0.80, "gf1t": 0.60, "corners": 4.3},
    "Olympique Marseille": {"gf": 2.10, "gc": 0.80, "gf1t": 1.00, "corners": 6.2},
    "AC Milan": {"gf": 1.70, "gc": 1.15, "gf1t": 0.75, "corners": 5.3},
    "Inter Milan": {"gf": 1.90, "gc": 0.85, "gf1t": 0.95, "corners": 5.1},
    "CD Olimpia": {"gf": 2.05, "gc": 0.70, "gf1t": 0.95, "corners": 5.2},
    "FC Motagua": {"gf": 1.70, "gc": 0.95, "gf1t": 0.75, "corners": 4.8},
    "CF Monterrey": {"gf": 1.75, "gc": 1.30, "gf1t": 0.80, "corners": 5.3}
}

col1, col2 = st.columns(2)

with col1:
    st.subheader("🏠 Local")
    equipo_loc = st.selectbox("Equipo Local", list(equipos_db.keys()), index=0)
    prio_loc = st.slider("Prioridad de Ganar (Local)", 1, 10, 8, key="p_loc")

with col2:
    st.subheader("✈️ Visita")
    equipo_vis = st.selectbox("Equipo Visita", list(equipos_db.keys()), index=1)
    prio_vis = st.slider("Prioridad de Ganar (Visita)", 1, 10, 7, key="p_vis")

ritmo = st.select_slider(
    "Ritmo / Intensidad del Partido",
    options=["Bajo (0.8x)", "Normal (1.0x)", "Alto (1.15x)", "Vertiginoso (1.25x)"],
    value="Alto (1.15x)"
)

# Factores de conversión
f_ritmo = float(ritmo.split("(")[1].replace("x)", ""))
f_prio_loc = 1.0 + (prio_loc - 5) * 0.04
f_prio_vis = 1.0 + (prio_vis - 5) * 0.04

if st.button("🚀 EJECUTAR SIMULACIÓN (10,000 PARTIDOS)", use_container_width=True):
    stats_loc = equipos_db[equipo_loc]
    stats_vis = equipos_db[equipo_vis]
    
    # Lambdas ajustados
    l_gf_loc = stats_loc["gf"] * 1.10 * f_prio_loc * f_ritmo
    l_gf_vis = stats_vis["gf"] * f_prio_vis * f_ritmo
    
    l_1t_loc = stats_loc["gf1t"] * f_prio_loc * f_ritmo
    l_1t_vis = stats_vis["gf1t"] * f_prio_vis * f_ritmo
    
    l_corn_tot = (stats_loc["corners"] + stats_vis["corners"]) * f_ritmo
    
    # Monte Carlo Engine (10,000 iteraciones)
    n = 10000
    g_loc = np.random.poisson(l_gf_loc, n)
    g_vis = np.random.poisson(l_gf_vis, n)
    
    g_1t_loc = np.random.poisson(l_1t_loc, n)
    g_1t_vis = np.random.poisson(l_1t_vis, n)
    
    corners_ft = np.random.poisson(l_corn_tot, n)
    corners_1t = np.random.poisson(l_corn_tot * 0.45, n)
    
    # Porcentajes calculados
    prob_loc = np.mean(g_loc > g_vis) * 100
    prob_empate = np.mean(g_loc == g_vis) * 100
    prob_vis = np.mean(g_loc < g_vis) * 100
    
    prob_1t_05 = np.mean((g_1t_loc + g_1t_vis) > 0.5) * 100
    prob_ft_25 = np.mean((g_loc + g_vis) > 2.5) * 100
    prob_corn_95 = np.mean(corners_ft > 9.5) * 100
    prob_corn_45_1t = np.mean(corners_1t > 4.5) * 100

    st.markdown("---")
    st.header("📊 Pronósticos Resultantes")
    
    m1, m2, m3 = st.columns(3)
    m1.metric("Gana Local", f"{prob_loc:.1f}%")
    m2.metric("Empate", f"{prob_empate:.1f}%")
    m3.metric("Gana Visita", f"{prob_vis:.1f}%")
    
    st.subheader("🎯 Líneas de Valor Sugeridas")
    st.write(f"- **Goles 1er Tiempo (> 0.5 1T):** {prob_1t_05:.1f}% de probabilidad ({np.mean(g_1t_loc + g_1t_vis):.2f} goles exp.)")
    st.write(f"- **Goles Partido Completo (> 2.5 FT):** {prob_ft_25:.1f}% de probabilidad ({np.mean(g_loc + g_vis):.2f} goles exp.)")
    st.write(f"- **Córners 1er Tiempo (> 4.5 1T):** {prob_corn_45_1t:.1f}% de probabilidad ({np.mean(corners_1t):.1f} córners exp.)")
    st.write(f"- **Córners Totales (> 9.5 FT):** {prob_corn_95:.1f}% de probabilidad ({np.mean(corners_ft):.1f} córners exp.)")
